program ::= statement_list EOF

statement_list ::= empty | statement_list statement

statement ::= variableDeclaration
            | functionDeclaration
            | expressionStatement
            | ifStatement
            | whileStatement
            | forStatement
            | returnStatement
            | block

block ::= "{" statement_list "}"

variableDeclaration ::= variableMutability optionalVariableType optionalVariableInitialization ";"

variableMutability ::= "let" | "var"

optionalVariableType ::= empty | ":" type

optionalVariableInitialization ::= empty | "=" expression

functionDeclaration ::= "fn" identifier "(" parameterList ")" "->" type block ;

parameterList ::= empty | parameter parameterListTail ;

parameterListTail ::= empty | "," parameter parameterListTail

parameter ::= identifier ":" type ;

expressionStatement ::= expression ";"

ifStatement ::= "if" expression block optionalElseBlock

optionalElseBlock ::= empty | "else" block

whileStatement ::= "while" expression block

forStatement ::= "for" forControl block ;

forControl ::= traditionalFor
             | iteratorFor ;

traditionalFor ::= optionalExpression ";" optionalExpression ";" optionalExpression ;

optionalExpression ::= empty | expression

iteratorFor ::= identifier "in" expression ;

returnStatement ::= "return" optionalExpression ";" ;

type ::= primitiveType
       | arrayType ;

primitiveType ::= "Int"
                | "Float"
                | "Bool"
                | "Char"
                | "String"
                | "Void"

arrayType
    ::= primitiveType "[" optional_int_literal "]" ;

optional_int_literal :== empty | int_literal

expression ::= assignment

assignment ::= logicalXor "=" assignment

logicalXor ::= logicalOr | logicalXor "xor" logicalOr

logicalOr ::= logicalAnd | logicalOr "or" logicalAnd

logicalAnd ::= equality | logicalAnd "and" equality

equality ::= comparison | equality ("==" | "!=") comparison

comparison ::= range | comparison ("<" | ">" | "<=" | ">=") range

range ::= term (".." term)? ;

term ::= factor | term ("+" | "-") factor

factor ::= power | factor ("*" | "/" | "%") power

power ::= cast | cast "^" power

cast ::= unary ("as" type)* ;

unary ::= primary | ("not" | "-" | "+") unary

primary ::= int_literal
          | float_literal
          | bool_literal
          | char_literal
          | string_literal
          | identifier
          | "(" expression ")"
          | functionCall

functionCall ::= identifier "(" argumentList ")"

argumentList ::= empty | primary argumentListTail

argumentListTail ::= empty | primary "," argumentListTail

bool_literal ::= "true"
               | "false"
